        function loadSectors() {

            const savedSectors = localStorage.getItem('bluevision_sectors');

            if (savedSectors) {

                sectors = JSON.parse(savedSectors);

            }

            const savedExpenses = localStorage.getItem('bluevision_expenses');

            if (savedExpenses) {

                expenses = JSON.parse(savedExpenses);

            localStorage.setItem('bluevision_sectors', JSON.stringify(sectors));

        function updateStats() {

            const totalSectors = sectors.length;

            const totalBushes = sectors.reduce((sum, s) => sum + (parseInt(s.quantity) || 0), 0);

            const totalProfit = sectors.reduce((sum, s) => sum + (parseFloat(s.profit_rub) || 0), 0);

            const potentialYield = Math.round(totalBushes * 2.5);

            const totalExpenses = expenses.reduce((sum, e) => sum + (e.cost_rub || 0), 0);


            document.getElementById('stat-sectors').textContent = totalSectors;

            document.getElementById('stat-bushes').textContent = totalBushes.toLocaleString('ru-RU');

            document.getElementById('stat-profit').textContent = totalProfit.toLocaleString('ru-RU');

        }

        function filterSectors() {

            const query = document.getElementById('search-input').value.toLowerCase().trim();

            if (!query) {

                renderSectors();

                return;

            }

            const filtered = sectors.filter(s => 

                (s.name && s.name.toLowerCase().includes(query)) || 

                (s.variety && s.variety.toLowerCase().includes(query))

            );

            renderSectors(filtered);

        }
